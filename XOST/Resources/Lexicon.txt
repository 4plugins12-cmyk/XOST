# XOST word engine vocabulary. Plain text so it is easy to extend.
#
#   type | phrases (comma separated, lower case) | fields separated by " ; "
#
# types:   cat     a sound category                    fields: cat=Pluck
#          kind    a specific sound family (recipes)   fields: kinds=Cat.recipe*weight ...
#          style   a genre / artist / producer         fields: label= tags= kinds= ops= use=
#          char    a character / mood word             fields: tags= kinds= ops= neg= use=
#          fx      an effect / movement word           fields: tags= ops= neg=
#          amount  scales the next word                fields: 0.5
#          neg     "no / without" - negates the next word(s)
#          refine  "more / less" - scales the next word and hints that you are tweaking the current sound
#          stop    words that are understood but carry no meaning ("a", "with", "type beat"...)
#
# A phrase ending in "+" (e.g. "darker+") also hints at a tweak of the current sound.
#
# ops (applied with the word's intensity k: 1 = normal, 0.5 = "a little", 1.6 = "very", -1 = "no", -0.6 = "less"):
#   id*f   multiply (f^k)        id+d   add (d*k)        id=v   set (k > 0 only)       id~v   move toward v
#   o*_    = every active oscillator (oa_ ob_ oc_)       f*_ = every active filter (f1_ f2_)
#   @lfo(dst,amount,division[,shape])  @mx(MatrixCategory)  @rk(MinirackCategory)  @chord(type[,voicing[,strum]])
#   @arp(mode,rate,octaves)  @oct(n)  @mono(glide ms)  @poly  @uni(n)  @pitchdrop(amount,ms)
# Turning up an effect's setting switches that effect on (starting from a gentle amount).
